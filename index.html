function configurarBaseDeDatos() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Estructura actualizada con los nuevos campos de Responsable 2 const estructura = [ { nombre: 'Usuarios', columnas: ['ID', 'Nombre', 'Correo', 'Password', 'Rol', 'Estado'] }, { nombre: 'alumnos', columnas: ['id', 'estatus', 'nombres', 'apellidos', 'fechaNacimiento', 'nivel', 'grado', 'grupo', 'tutor', 'telefono', 'responsable2', 'telefonoResp2', 'tipoCuota', 'tipoSangre', 'alergias', 'padecimientos', 'contactoEmergencia', 'telefonoEmergencia'] }, { nombre: 'maestros', columnas: ['id', 'nombres', 'apellidos', 'telefono', 'email', 'niveles', 'rfc', 'curp'] }, { nombre: 'cuotas', columnas: ['id', 'nombre', 'costo'] }, { nombre: 'pagos', columnas: ['id', 'alumnoId', 'fecha', 'conceptos', 'metodosDePago', 'total', 'notas', 'estado', 'motivoAnulacion'] }, { nombre: 'abonos', columnas: ['id', 'alumnoId', 'monto', 'metodo', 'fechaAbono'] }, { nombre: 'cajaChica', columnas: ['id', 'fecha', 'tipo', 'concepto', 'monto'] }, { nombre: 'nomina', columnas: ['id', 'maestroId', 'maestroNombre', 'fechaPago', 'monto'] } ]; const nombresRequeridos = estructura.map(e => e.nombre); const hojasExistentes = ss.getSheets(); // 1. LIMPIEZA: Borrar las hojas que no están en la estructura nueva hojasExistentes.forEach(hoja => { const nombre = hoja.getName(); if (!nombresRequeridos.includes(nombre)) { if (ss.getSheets().length > 1) { // Prevenir error de borrar todas las hojas ss.deleteSheet(hoja); } } }); // 2. CREACIÓN/ACTUALIZACIÓN: Actualizar encabezados estructura.forEach(hojaDef => { let sheet = ss.getSheetByName(hojaDef.nombre); if (!sheet) { sheet = ss.insertSheet(hojaDef.nombre); } // Poner o actualizar los encabezados en la Fila 1 (No borra el contenido de abajo) const range = sheet.getRange(1, 1, 1, hojaDef.columnas.length); range.setValues([hojaDef.columnas]); // Estilo institucional range.setFontWeight("bold").setBackground("#1e3a8a").setFontColor("#ffffff"); sheet.setFrozenRows(1); sheet.autoResizeColumns(1, hojaDef.columnas.length); }); // 3. USUARIO ADMIN DE RESERVA (Solo lo crea si la hoja está vacía) const sheetUsuarios = ss.getSheetByName('Usuarios'); if (sheetUsuarios && sheetUsuarios.getLastRow() === 1) { sheetUsuarios.appendRow(['1', 'Administrador', 'admin@escuela.mx', '1234', 'Admin', 'Activo']); } SpreadsheetApp.getUi().alert("¡Éxito! Tu BD se ha actualizado correctamente. Se agregaron las columnas para Responsable 2 (Deudor)."); }
